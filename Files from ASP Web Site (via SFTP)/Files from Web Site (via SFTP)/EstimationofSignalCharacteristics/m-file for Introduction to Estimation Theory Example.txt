%% Set up workspace

clear
close all

set(0,'defaultaxesfontsize',22);
set(0,'defaulttextfontsize',22);
linwidth = 2;

%% Sample Mean and Variance

% Generate 2000 examples of a three-dimensional vector having unit mean and
% variance.

X = randn(3,2000) + ones(3,2000);

% Estimate the mean (average) of each of the 2000 examples.
mean_est = (1/3)*[1 1 1]*X;

% Estimate the variance using the biased sample variance estimator
var_bias = (1/3)*diag((X-[1 1 1]'*mean_est)'*(X-[1 1 1]'*mean_est));

% The unbiased estimator of variance
var_unbias = (3/2)*var_bias;

% Display histograms of results

figure
hist(mean_est,[-2:.1:4]);
xlim([-2,4])
hold on
h=plot([1 1],[0 200],'-r');
set(h,'LineWidth',linwidth);
h = get(0,'CurrentFigure');
set(h,'Position',[160,180,700,500])
xlabel('Estimated Mean')
ylabel('Number of Cases')
title('Sample Mean Estimator')

figure
hist(var_bias,[0:.1:7])
xlim([-.2,7])
hold on
h=plot([1 1],[0 300],'-r');
set(h,'LineWidth',linwidth);
h = get(0,'CurrentFigure');
set(h,'Position',[160,180,700,500])
xlabel('Estimated Variance')
ylabel('Number of Cases')
title('Biased Variance Estimator')

figure
hist(var_unbias,[0:.1:10])
xlim([-.2,7])
hold on
h=plot([1 1],[0 250],'-r');
set(h,'LineWidth',linwidth);
h = get(0,'CurrentFigure');
set(h,'Position',[160,180,700,500])
xlabel('Estimated Variance')
ylabel('Number of Cases')
title('Unbiased Variance Estimator')
