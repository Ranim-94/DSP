%% Power Spectrum Estimation Examples: Welch's Method
%
% Copyright 2014 Barry Van Veen

%% Initialize Workspace

clear
close all

set(0,'defaultaxesfontsize',22);
set(0,'defaulttextfontsize',22);
linwidth = 1.5;

fs = 100; % Assume 100 Hz sampling frequency for illustration

%% Random Process Construction
%
% Use an FIR filter applied to white noise to construct random signals with
% a known power spectrum.

freqs = [0 .2 .21 .23 .24 .25 .26 .27 .28 .52 .525 1];
bandgain = [0.1 0.1 8 10 0 0 10 8 .03 .03 .5 .5];
wghts = [ 10 1 100 1 100 2];

B = firpm(700,freqs,bandgain,wghts);

[H,w1] = freqz(B,1,2048,fs);
figure
h=plot(w1,20*log10(abs(H)));
set(h,'LineWidth',linwidth);
xlabel('Frequency (Hz)')
title('True Power Spectrum')
ylabel('dB')

figure
h=plot(w1,(abs(H)));
set(h,'LineWidth',linwidth);
xlabel('Frequency (Hz)')
title('True Power Spectrum')
ylabel('Watts')

% Generate 500 random processes with this power spectrum

x = randn(500,1725);
for i = 1:500
    z(i,:) = filter(B,1,x(i,:));
    % select 1024 points after initial conditions have decayed
    y(i,:) = z(i,701:1724);  
end

%% Compute Power Spectra
%
% Compute for each of the 500 realizations of the process so that we can
% estimate the mean power spectrum

for i = 1:500
    [Py1024(i,:),w] = pwelch(y(i,:),1024,0,2048,fs);
    [Py1024rect(i,:),w] = pwelch(y(i,:),ones(1,1024),0,2048,fs);
    [Py256(i,:),w] = pwelch(y(i,:),256,0,2048,fs);
    [Py256rect(i,:),w] = pwelch(y(i,:),ones(1,256),0,2048,fs);
    [Py256ovrlp5(i,:),w] = pwelch(y(i,:),256,64,2048,fs);
    [Py256ovrlp16(i,:),w] = pwelch(y(i,:),256,208,2048,fs);
    [Py128(i,:),w] = pwelch(y(i,:),128,0,2048,fs);
    [Py64(i,:),w] = pwelch(y(i,:),64,0,2048,fs);
end

figure
for i = 1:3
    subplot(3,1,i)
    h = plot(w1,10*log10(abs(H).^2),'r',w,10*log10((fs/2)*Py1024(i,:)),'b'); 
    %mult by fs/2 to convert power spectral density to power spectrum since
    %in this case we are looking at the one-sided density.  If two-sided,
    %then we multiply by fs.
    set(h,'LineWidth',linwidth);
    ylim([-60,25])
end
%xlabel('Frequency rads/sec')
subplot(3,1,1)
legend('true','Periodogram')
h = get(0,'CurrentFigure');
set(h,'Position',[160,180,700,500])

figure
for i = 1:3
    subplot(3,1,i)
    h = plot(w1,10*log10(abs(H).^2), 'r',w,10*log10((fs/2)*Py256(i,:)),'b');
    set(h,'LineWidth',linwidth);
    ylim([-60,25])
end
subplot(3,1,1)
legend('true','welch 4')
h = get(0,'CurrentFigure');
set(h,'Position',[160,180,700,500])

figure
for i = 1:3
    subplot(3,1,i)
    h = plot(w1,10*log10(abs(H).^2), 'r',w,10*log10((fs/2)*Py256ovrlp5(i,:)),'b');
    set(h,'LineWidth',linwidth);
    ylim([-60,25])
end
subplot(3,1,1)

end

    set(h,'LineWidth',linwidth);
    ylim([-60,25])
end
subplot(3,1,1)

    set(h,'LineWidth',linwidth);

end
