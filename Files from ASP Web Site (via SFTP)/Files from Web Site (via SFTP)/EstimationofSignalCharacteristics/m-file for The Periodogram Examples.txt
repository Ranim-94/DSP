%% Initialize Workspace

clear
close all

set(0,'defaultaxesfontsize',22);
set(0,'defaulttextfontsize',22);
linwidth = 2;

%% Periodogram estimates
%
% Generate data with a known power spectrum by passing white noise through
% a moving-average system. Show that the variance is proportional to
% the square of the true power spectrum, independent of data length.

[H,om] = freqz([.5 .5],1,1024); % System frequency response

% Compute periodogram using L = 64 data values. Show 3 examples.
figure
for i = 1:3    
    w = randn(1,80);
    x = filter([0.5 0.5],1,w); % Moving average data
    X = abs(fft(x(10:73),1024)).^2;
    % .^2 raises each element of the vector to the 2 power
    
    subplot(3,1,i)
    h=plot([0:511],X(1:512)/64,'b',om*1024/(2*pi),abs(H).^2,'r'); 
    set(h,'LineWidth',linwidth);
    xlim([0 511])
    ylim([0 5])
end
subplot(311)
legend('Periodogram','True')
h = get(0,'CurrentFigure');
set(h,'Position',[160,180,650,500])
