% Copyright 2014 Barry Van Veen
%
% Generate example scaling functions from Daubechies' wavelets.

%% Set Up Workspace

clear
close all

set(0,'defaultaxesfontsize',18);
set(0,'defaulttextfontsize',18);
linwidth = 2;

%% Wavelets 
%
% Different vanishing moments

L = [8 8 16 32 64 128 256 512 1024];

C = [ zeros(1,22) 1 zeros(1,1024-23)];

Daub1 = waverec(C,L,'db1');

Daub3 = waverec(C,L,'db3');

Daub8 = waverec(C,L,'db8');

figure
h = plot(0:1023,Daub1, 1200:2223, Daub3,'b', 2400:3423,Daub8,'b');
set(h,'LineWidth',linwidth)

%% Illustrate support-vanishing moment tradeoff
%

lineup = [0:200]/40;

linedown = [100:-1:0]/20;

quadup = 5*([0:200]/200).^2;

quaddown = -5*([0:150]/150).^2 + [200:-1:50]/50;

sig = [zeros(1,50), -0.5*ones(1,125), lineup, linedown, quadup,... 
    4+linedown/5, quaddown, zeros(1,1024-929)];

figure
h = plot(sig);
ylim([-6,6])
set(h,'LineWidth',linwidth)

% Use only one vanishing moment

wname = 'db1';
[W,P] = wavedec(sig,3,wname);

N = length(sig);

c1 = W(1:P(1));
d1 = W(P(1)+1:P(1)+P(2));
d2 = W(P(1)+P(2)+1:P(1)+P(2)+P(3));
d3 = W(P(1)+P(2)+P(3)+1:P(1)+P(2)+P(3)+P(4));

% Display results
figure
    subplot(3,2,2); 
    h=stem(0:P(4)-1,d3,'b');
    xlim([0,P(4)-1])
    set(h,'LineWidth',linwidth)
    
    subplot(3,2,4); 
    h=stem(0:P(3)-1,d2,'b'); 
     xlim([0,P(3)-1])
     set(h,'LineWidth',linwidth)
  
    subplot(3,2,6); 
    h=stem(0:P(2)-1,d1,'b'); 
    xlim([0,P(2)-1])
    set(h,'LineWidth',linwidth)
    
    subplot(3,2,5); 
    h=stem(0:P(1)-1,c1,'b'); 
    xlim([0,P(1)-1])
    set(h,'LineWidth',linwidth)
    
    subplot(3,2,1);
    h = plot(0:N-1,sig,'r');
    xlim([0,N-1])
    ylim([-5,5.5])
    set(h,'LineWidth',linwidth)
    
    h = get(0,'CurrentFigure');
set(h,'Position',[0,0,1500,750])
