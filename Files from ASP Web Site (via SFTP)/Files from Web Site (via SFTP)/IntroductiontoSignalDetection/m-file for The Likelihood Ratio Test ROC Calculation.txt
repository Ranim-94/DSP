%% The Likelihood Ratio Test ROC
%
% Copyright 2014 Barry Van Veen
%
% Known signal in Gaussian noise detection performance
%
% Signal s, white noise var sigma^2, SNR = s^T*s/sigma^2

%% Initialize Workspace

clear
close all

set(0,'defaultaxesfontsize',22);
set(0,'defaulttextfontsize',22);
linwidth = 1.5;

%% Calculate and Display ROC

SNR = [ 0.1 1 3 10 ]; %range of SNRs to evaluate

thresh = [-5:.02:8]; %trace out ROC using a wide range of thresholds

for i=1:length(thresh)
    % Pfa depends only on the threshold and noise variance
    Pfa(i) = 1 - normcdf(thresh(i),0,1);
    
    for j = 1:length(SNR)
        % Pd depends on threshold, noise variance, and SNR
        Pd(i,j) = 1-normcdf(thresh(i),sqrt(SNR(j)),1);
    end
end

figure
h = plot(Pfa,Pd(:,1),'k',Pfa,Pd(:,2),'b',Pfa,Pd(:,3),'g',Pfa,Pd(:,4),'r');
set(h,'LineWidth',linwidth);
xlabel('Prob False Alarm')
ylabel('Prob Detection')
title('Receiver Operating Characteristic')
legend(['SNR = ',num2str(SNR(1))],['SNR = ',num2str(SNR(2))],['SNR = ',num2str(SNR(3))],['SNR = ',num2str(SNR(4))])

h = get(0,'CurrentFigure');
set(h,'Position',[160,180,700,500])
