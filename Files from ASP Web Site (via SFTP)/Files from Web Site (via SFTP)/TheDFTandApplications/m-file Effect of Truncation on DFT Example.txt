%% Effect of truncation on the DTFT
%
% Copyright 2014 Barry Van Veen
%
% Evaluate DFT approximation of the FT using rectangular window and two
% sinusoid of different amplitudes and frequencies.

close all
clear

set(0,'defaultaxesfontsize',22);
set(0,'defaulttextfontsize',22);
linwidth = 2;

%% DTFT of rectangular window
%
% Compare magnitudes of of DTFT of window for different lengths, N = 25 and
% N = 50.  Note that mainlobe is narrower for N=50, but ratio of mainlobe
% height to sidelobe height is unchanged.

Om = [-1000:1000]*pi/1000; % values in frequency to eval W(Om)

N = 25;
W1 = exp(j*Om*(N-1)/2).*sin(Om*N/2)./sin(Om/2);  % ./ does element by element divide

N = 50;
W2 = exp(j*Om*(N-1)/2).*sin(Om*N/2)./sin(Om/2);  

figure(1), 
subplot(2,2,1)
h = plot(Om,abs(W1));
set(h,'LineWidth',linwidth);
title('N = 25, linear scale')
xlabel('Freq (rads)')
axis([-pi,pi,0,max(abs(W1))])

subplot(2,2,3)
h=plot(Om,abs(W2));
set(h,'LineWidth',linwidth);
title('N = 50, linear scale')
xlabel('Freq (rads)')
axis([-pi,pi,0,max(abs(W2))])

subplot(2,2,2)
h=plot(Om,20*log10(abs(W1)));
set(h,'LineWidth',linwidth);
title('N = 25, dB scale')
xlabel('Freq (rads)')
ylabel('dB')
axis([-pi,pi,-30,max(20*log10(abs(W1)))])

subplot(2,2,4)
h=plot(Om,20*log10(abs(W2)));
set(h,'LineWidth',linwidth);
title('N = 50, dB scale')
xlabel('Freq (rads)')
ylabel('dB')
axis([-pi,pi,-30,max(20*log10(abs(W2)))])

h = get(0,'CurrentFigure');
set(h,'Position',[160,180,1920,1080])


%% DTFT of truncated cosines closely spaced in frequency
%
% One cosine at 7*pi/100 the other at 7.5*pi/100.  Width of the mainlobe
% for the rectangular window is 4*pi/100, much larger than the separation
% between the two cosines.

N = 100;

x1 = cos([0:N-1]*7*pi/100);
x2 = cos([0:N-1]*7.5*pi/100);

% Use DFT with very large number of freq samples to approx DTFT

X1 = fft(x1,4096);
X2 = fft(x2,4096);
Xsum = fft(x1+x2,4096);

% Display DTFT approx(only positive freqs up to pi/4)

figure(2)
subplot(3,2,1)
h=plot([0:511]*pi/2048,(abs(X1(1:512))))
set(h,'LineWidth',linwidth);
title('Cos with freq 7*pi/100')
xlabel('freq rads')

subplot(3,2,2)
h=plot([0:511]*pi/2048,20*log10(abs(X1(1:512))));
set(h,'LineWidth',linwidth);
title('Cos with freq 7*pi/100')
xlabel('freq rads')
ylabel('dB')
axis([0 pi/4 -20 40])

subplot(3,2,3)
h=plot([0:511]*pi/2048,(abs(X2(1:512))));
set(h,'LineWidth',linwidth);
title('Cos with freq 7.5*pi/100')
xlabel('freq rads')

subplot(3,2,4)
h=plot([0:511]*pi/2048,20*log10(abs(X2(1:512))));
set(h,'LineWidth',linwidth);
title('Cos with freq 7.5*pi/100')
xlabel('freq rads')
ylabel('dB')

axis([0 pi/4 -20 40])
subplot(3,2,5)
h=plot([0:511]*pi/2048,(abs(Xsum(1:512))));
set(h,'LineWidth',linwidth);
title('Sum of Cos freqs 7pi/100 and 7.5pi/100')
xlabel('freq rads')

subplot(3,2,6)
h=plot([0:511]*pi/2048,20*log10(abs(Xsum(1:512))));
set(h,'LineWidth',linwidth);
title('Sum of Cos freqs 7pi/100 and 7.5pi/100')
xlabel('freq rads')
ylabel('dB')
axis([0 pi/4 -20 40])

h = get(0,'CurrentFigure');
set(h,'Position',[160,180,1920,1080])

%% DFT for large amplitude and small amplitude cosines
