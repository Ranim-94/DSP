%% Exponential, Step, and Impulse Signals%% Copyright Barry Van Veen 2014 %% Set Up Workspace close allclear % Make graphics visible for large plots load Guitar_G set(0,'defaultaxesfontsize',24);set(0,'defaulttextfontsize',24);linwidth = 2; %% Guitar Envelope fs = 44100;  % CD sampling rate audiowrite('Guitar_ G.wav',[ zeros(20000,1);G],44100) t_tot = [0:length(G)-1]/fs; % First we find the envelope of the note G y = decimate(abs(G),10);  %4410 Hz Sampling ratey = decimate(abs(y),10);  %441 Hz Sampling ratey = decimate(abs(y),10);  %44.1 Hz Sampling rate [Bfilt,Afilt] = butter(6,0.75);env = filtfilt(Bfilt,Afilt,abs(y)); t_env = [0:length(env)-1]/(fs/1000); figure(1)subplot(211)hp = plot(t_tot,G,t_env,2.3*env,'r--');  % 2.3*env accounts for avg value of envelope as determined by low pass % filtering the absolute value relative to the actual peaks of G - this% factor was determined empirically by looking at the two waveforms set(hp,'LineWidth',linwidth)xlabel('Time (sec)')ylabel('Norm Amp.')title('Guitar Note G')legend('Guitar','Envelope') subplot(212)hp = plot(t_tot(88201:92612),G(88201:92612));set(hp,'LineWidth',linwidth);xlabel('Time (sec)')ylabel('Norm. Amp.')title('Section of Guitar Note G')axis([t_tot(88201),t_tot(92612),-0.1,0.1]) h = get(0,'CurrentFigure');% a = get(h,'Position');set(h,'Position',[0,0,2560,1440]) pause % play sound sound([zeros(1,20000),G'],fs) % deadspace at start of sound to eliminate auditory artifact in my sound% card output pause sound([zeros(1,20000),G'],fs)    % Now fit x[n] = a*b^n to the envelope by finding a, b.  Use fact that % x[n+1]/x[n] = b. Average this ratio over the tail of env. % Fit env with two exponentials: 1 for samples 22:75, then 76:end % First exponential b1 = sum(env(23:75)./env(22:74))/length(env(23:75)); exp1 = b1.^[1:length(env)]; % Now get a1 using least squares a1 = exp1(22:75)*env(22:75)/(exp1(22:75)*exp1(22:75)'); exp1 = a1*exp1; % Second exponential b2 = sum(env(76:end)./env(75:end-1))/length(env(76:end)); exp2 = b2.^[1:length(env)]; a2 = exp2(76:end)*env(76:end)/(exp2(76:end)*exp2(76:end)'); exp2 = a2*exp2; % Display results figure(2)hp = plot(t_tot,G,'g-',t_env(1:110),2.3*exp1(1:110),'r--',t_env(50:end),2.3*exp2(50:end),'b--');  % As above, 2.3*env accounts for avg value of envelope relative to peaks of G set(hp,'LineWidth',linwidth)xlabel('Time (sec)')ylabel('Norm Amp.')title('Guitar Note G')legend('Guitar','Exponential 1','Exponential 2') h = get(0,'CurrentFigure');% a = get(h,'Position');set(h,'Position',[0,0,2560,1440]) pause % Now we'll approximate the G note using the exponential model for the% envelope and a sinusoid of the appropriate frequency. Exp_env = [ zeros(1,21000), 2.3*a1*b1.^([21001:75000]/1000), 2.3*a2*b2.^([86000:length(exp2)*1000]/1000)]; % Fundamental frequency of G is 196 Hz t_syn = [0:length(Exp_env)-1]/fs; G_syn = Exp_env.*sin(2*pi*196*t_syn); figure(3)subplot(311)hp = plot(t_syn,G_syn,t_syn,Exp_env,'r--');  % 2.3*env accounts for avg value of envelope as determined by low pass % filtering the absolute value relative to the actual peaks of G - this% factor was determined empirically by looking at the two waveforms set(hp,'LineWidth',linwidth)xlabel('Time (sec)')ylabel('Norm Amp.')title('Synthetic G')legend('Synthetic Note','Envelope') subplot(312)hp = plot([88201:92612]/44100,G_syn(88201:92612));set(hp,'LineWidth',linwidth);xlabel('Time (sec)')ylabel('Norm. Amp.')title('Section of Synthesized Note G')axis([t_tot(88201),t_tot(92612),-0.1,0.1]) subplot(313)hp = plot(t_tot(88201:92612),G(88201:92612));set(hp,'LineWidth',linwidth);xlabel('Time (sec)')ylabel('Norm. Amp.')title('Section of Guitar Note G')axis([t_tot(88201),t_tot(92612),-0.1,0.1]) h = get(0,'CurrentFigure');% a = get(h,'Position');set(h,'Position',[0,0,2560,1440]) pause % play Original sound sound([zeros(1,20000),G'],fs) % deadspace at start of sound to eliminate auditory artifact in my sound% card output pause %Play synthesized sound sound(2*G_syn,fs) audiowrite('Synthesized_G.wav',[ zeros(20000,1);G_syn'],44100)
