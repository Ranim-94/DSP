%% Sinusoidal Signals Examples%% Copyright Barry Van Veen 2014 %% Set up Workspace close allclear % Make graphics visible for large plots set(0,'defaultaxesfontsize',22);set(0,'defaulttextfontsize',22);linwidth = 2; %% C-major scale fs = 22050;  % one-half CD sampling rate % Frequencies of notes in C-major scale C = 523.28;D = 587.36;E = 659.28;F = 698.48;G = 784;A = 880;B = 987.84; % Construct scale with 1-second notes k = 0:22049;  % 1 sec at 22050 samples per second  % window to turn on and turn off tone smoothly (0.2 sec transition) w = [ sin(pi*[1:4410]/8820), ones(1,22050-8820), cos(pi*[1:4410]/8820)]; scale = [ w.*sin(2*pi*C*k/fs), w.*sin(2*pi*D*k/fs), w.*sin(2*pi*E*k/fs), w.*sin(2*pi*F*k/fs), ...    w.*sin(2*pi*G*k/fs), w.*sin(2*pi*A*k/fs), w.*sin(2*pi*B*k/fs), w.*sin(4*pi*C*k/fs) ];  t = [0:440]/fs; % 0.1 sec long time axis t_tot = [0:length(scale)-1]/fs; figure(1)subplot(311)hp = plot(t_tot,scale);set(hp,'LineWidth',linwidth)xlabel('Time (sec)')ylabel('Norm Amp.')title('Scale: C, D, E, F, G, A, B, C') subplot(312)hp = plot(1000*t,sin(2*pi*C*t));set(hp,'LineWidth',linwidth);xlabel('Time (ms)')ylabel('Norm. Amp.')title('Section of note C: 523.28 Hz') subplot(313)hp = plot(1000*t,sin(2*pi*A*t));set(hp,'LineWidth',linwidth);xlabel('Time (ms)')ylabel('Norm. Amp.');title('Section of note A: 880 Hz') h = get(0,'CurrentFigure');% a = get(h,'Position');set(h,'Position',[0,0,2560,1440]) pause % play sound sound([zeros(1,20000),0.8*scale],fs) % amplitude 0.8 to avoid clipping sound in output% deadspace at start of sound to eliminate auditory artifact in my sound% card output audiowrite('music_scale.wav',[zeros(1,20000),0.8*scale],fs) %% Saxophone Note % Now construct saxophone tone as sum of sinusoidal components load SaxA_note  A_per = Sax_A(21125:21224); %extract one period of the sound A_frq = fft(A_per); %decompose signal into frequency components % Use first n_comp components to reconstruct signal from sinusoids n_comp = 8;disp_comp = [1:3,7,8]; % components to display A_comp = zeros(n_comp,100); % begin with first component, then accumulate the restA_comp(1,:) = abs(A_frq(2))/50*cos(2*pi*[0:99]/100 + phase(A_frq(2))); for i=2:n_comp    A_comp(i,:) = A_comp(i-1,:) + abs(A_frq(i+1))/50*cos(2*pi*i*[0:99]/100 + phase(A_frq(i+1)));end % string 10 periods together for display A_tot = A_per'; %original soundA_part = A_comp; %component representationfor i=1:9    A_tot = [ A_tot, A_per']; %concatenate periods of original sound    A_part = [A_part, A_comp]; %concatenate periods of component representationend % display 1000 samplest_sub = 1000*[0:999]/44100; ax = [t_sub(1), t_sub(end), -0.4, 0.4]; figure(2)subplot(211)hp = plot(t_sub,A_tot);set(hp,'LineWidth',linwidth);xlabel('Time (ms)')ylabel('Norm. Amp.')title('Original Saxophone A')axis(ax) subplot(212)hp = plot(t_sub,A_part(1,:));set(hp,'LineWidth',linwidth);xlabel('Time (ms)')ylabel('Norm. Amp.')title('First Sinusoidal Component')axis(ax) h = get(0,'CurrentFigure');set(h,'Position',[0,0,1280,720]) pause for i = 2:length(disp_comp)    subplot(311)    hp = plot(t_sub,A_part(disp_comp(i)-1,:));    set(hp,'LineWidth',linwidth);    xlabel('Time (ms)')    ylabel('Norm. Amp.')    title(['Sum of ',num2str(disp_comp(i)-1),' Components'])    axis(ax)         subplot(312)    hp = plot(t_sub,A_part(disp_comp(i),:)-A_part(disp_comp(i)-1,:));    set(hp,'LineWidth',linwidth);    xlabel('Time (ms)')    ylabel('Norm. Amp.')    title([num2str(disp_comp(i)),'-th Component'])    axis(ax)         subplot(313)    hp = plot(t_sub,A_part(disp_comp(i),:));    set(hp,'LineWidth',linwidth);    xlabel('Time (ms)')    ylabel('Norm. Amp.')    title(['Sum of ',num2str(disp_comp(i)),' Components'])    axis(ax)     pauseend subplot(211)hp = plot(t_sub,A_tot);set(hp,'LineWidth',linwidth);xlabel('Time (ms)')ylabel('Norm. Amp.')title('Original Saxophone A')axis(ax) subplot(212)hp = plot(t_sub,A_part(n_comp,:));set(hp,'LineWidth',linwidth);xlabel('Time (ms)')ylabel('Norm. Amp.')title(['Approx. Using ',num2str(n_comp),' Sinusoids'])axis(ax) pause  % Now play original and approximated sax sound%% Construct approximately 1 second length of approximate sax A audiowrite('SaxA.wav',[ zeros(20000,1);Sax_A(20000:80000)],44100) for k = 1:length(disp_comp)        A_approx = A_part(disp_comp(k),:);     for i = 1:60        A_approx = [A_approx, A_part(disp_comp(k),:)];    end        name = ['SaxApprox_',num2str(disp_comp(k)),'comps.wav'];        audiowrite(name,[zeros(1,20000),A_approx],44100)end
