%% Prepare Workspace

clear
close all
clc
load Moon.mat % loads image named 'MadisonMoon'

im = double(MadisonMoon); % convert from uint8 to double precision format

% Display image

figure
imagesc(im)
axis equal
axis tight
colormap(gray)
title('Original Image')

%% Problem 1a
%

h_a = ones(6,6)/36;
ima = conv2(im,h_a);

figure
imagesc(ima)
axis equal
axis tight
colormap(gray)
title('6 by 6 smoothing window')

%% Problem 1b
%

h_b = ??;
imb = conv2(im,??);

figure
imagesc(imb)
axis equal
axis tight
colormap(gray)
title('16 by 16 smoothing window')

%% Problem 1c
%

h_h = ones(1,6)/6;
h_v = h_h';

im_h = conv2(im,h_h);
im_hv = conv2(im_h,??);

figure
imagesc(im_hv)
axis equal
axis tight
colormap(gray)
title('Horiz then Vertically Blurred Image')

norm(ima-im_hv,'fro')

figure
image(ima-im_hv)
axis equal
axis tight
colormap(gray)
title('Difference of horiz/vert-blurred image and part a')

% Note that the image command was used instead of imagesc.  Imagesc is
% deceptive in this case since the differences between imb and imcc are on
% the order of 10^{-14}.
