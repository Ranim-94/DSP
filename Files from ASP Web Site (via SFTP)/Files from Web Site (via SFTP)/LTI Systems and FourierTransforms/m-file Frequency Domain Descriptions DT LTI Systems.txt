%% Frequency-Domain Descriptions for Discrete-Time Linear Time-Invariant Systems
%
% Barry Van Veen  Copyright 2015
%
%% Set up workspace

close all
clear

set(0,'defaultaxesfontsize',22);
set(0,'defaulttextfontsize',22);
linwidth = 2;

%% First-Order System
%

b = 0.2;
a = [ 1 -0.8];

[H,W] = freqz(b,a,2048);

figure
g = plot(W,abs(H));
set(g,'LineWidth',linwidth);
xlabel('Frequency rads  ')
xlim([0,pi])
ylabel('|H(e^{j\omega})|')

figure
g = plot(W,phase(H));
set(g,'LineWidth',linwidth);
xlabel('Frequency rads/sec  ')
xlim([0,pi])
ylabel('arg\{H(e^{j\omega})\} rads')
